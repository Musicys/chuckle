### 2.2 类型定义 (`types.ts`)

```typescript
// 菜单项类型
export interface MenuItem {
   id: string; // 唯一标识
   name: string; // 显示名称
   icon: string; // 图标名称（Element Plus Icons）
   handler: () => void; // 点击处理函数
   visible?: boolean; // 是否显示
   divider?: boolean; // 是否为分隔线
}

// 菜单分组类型
export interface MenuGroup {
   id: string;
   items: MenuItem[];
}

// 右键菜单状态
export interface RightMenuState {
   visible: boolean;
   x: number;
   y: number;
}
```

### 2.3 组件设计 (`index.vue`)

```vue
<template>
   <Teleport to="body">
      <!-- 遮罩层 -->
      <div
         v-if="visible"
         class="right-menu-mask"
         @click="hideMenu"
         @contextmenu.prevent></div>

      <!-- 右键菜单 -->
      <Transition name="fade">
         <div
            v-if="visible"
            class="right-menu"
            :style="{ top: `${y}px`, left: `${x}px` }"
            @contextmenu.prevent>
            <template v-for="group in menuGroups" :key="group.id">
               <div v-for="item in group.items" :key="item.id">
                  <!-- 分隔线 -->
                  <div v-if="item.divider" class="menu-divider"></div>

                  <!-- 菜单项 -->
                  <div
                     v-else-if="item.visible !== false"
                     class="menu-item"
                     @click="handleClick(item)">
                     <el-icon :size="16">{{ getIcon(item.icon) }}</el-icon>
                     <span>{{ item.name }}</span>
                  </div>
               </div>
            </template>
         </div>
      </Transition>
   </Teleport>
</template>
```

### 2.4 逻辑 Hook (`useRightMenu.ts`)

```typescript
import { ref, onMounted, onUnmounted } from 'vue';
import type { MenuItem, RightMenuState } from './types';

export function useRightMenu() {
   const state = ref<RightMenuState>({
      visible: false,
      x: 0,
      y: 0
   });

   // 菜单项配置
   const menuItems = ref<MenuItem[]>([]);

   // 显示菜单
   const showMenu = (e: MouseEvent) => {
      e.preventDefault();

      // 获取菜单位置
      let x = e.clientX;
      let y = e.clientY;

      // 边界检测
      const menuWidth = 180;
      const menuHeight = 400;

      if (x + menuWidth > window.innerWidth) {
         x = window.innerWidth - menuWidth;
      }
      if (y + menuHeight > window.innerHeight) {
         y = window.innerHeight - menuHeight;
      }

      state.value = { visible: true, x, y };
   };

   // 隐藏菜单
   const hideMenu = () => {
      state.value.visible = false;
   };

   // 点击菜单项
   const handleItemClick = (item: MenuItem) => {
      hideMenu();
      item.handler();
   };

   // 注册事件监听
   onMounted(() => {
      document.addEventListener('contextmenu', showMenu);
      document.addEventListener('click', hideMenu);
   });

   onUnmounted(() => {
      document.removeEventListener('contextmenu', showMenu);
      document.removeEventListener('click', hideMenu);
   });

   return {
      state,
      menuItems,
      showMenu,
      hideMenu,
      handleItemClick
   };
}
```

## 三、开发思路

### 3.1 核心实现步骤

#### 步骤1：创建类型定义文件

- 定义 `MenuItem`、`MenuGroup`、`RightMenuState` 接口
- 确保类型安全和代码可维护性

#### 步骤2：创建样式文件

- 使用 SCSS 编写菜单样式
- 实现圆角、阴影、悬停效果
- 添加过渡动画

#### 步骤3：创建右键菜单组件

- 使用 `Teleport` 挂载到 `body`
- 实现位置计算和边界检测
- 使用 `v-for` 动态渲染菜单项
- 添加遮罩层防止点击穿透

#### 步骤4：实现功能逻辑

- 导航操作：使用 Vue Router 进行页面跳转
- 文本操作：使用 Clipboard API 实现复制功能
- 主题切换：调用现有主题切换方法
- 搜索功能：打开搜索弹窗

#### 步骤5：注册全局组件

- 在 `App.vue` 中注册右键菜单组件
- 确保全局生效

```typescript
{
  id: 'copy',
  name: '复制文本',
  icon: 'Copy',
  handler: async () => {
    const selectedText = window.getSelection()?.toString();
    if (selectedText) {
      await navigator.clipboard.writeText(selectedText);
      ElMessage.success('复制成功');
    }
  },
  visible: window.getSelection()?.toString().length > 0
}
```

### 3.3 交互优化

1. **边界检测**：确保菜单不会超出屏幕边界
2. **点击外部关闭**：点击菜单外区域自动关闭
3. **事件阻止**：阻止默认右键菜单和事件冒泡
4. **平滑过渡**：添加显示/隐藏动画效果
5. **响应式设计**：移动端可选择禁用右键菜单

## 六、注意事项

1. **移动端兼容性**：移动端通常长按触发右键菜单，建议在小屏幕设备上禁用自定义右键菜单
2. **权限问题**：复制功能需要用户交互权限，确保在用户触发的事件中调用
3. **性能优化**：使用 `Teleport` 避免重新渲染问题，使用 `computed` 优化菜单配置
4. **样式一致性**：使用项目现有的 CSS 变量保持样式统一
5. **图标引入**：使用 Element Plus Icons 保持图标风格一致
